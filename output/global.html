Example
const paintConfig = {};
fadeIn(paintConfig);
slide(paintConfig, 0, 100, 0, 50, 30);

  {
    config: paintConfig,
    handler(ctx, config) {
      ctx.globalAlpha = config.alpha as number;
      ctx.fillRect(config.x as number, config.y as number, 10, 10);
    },
  },
